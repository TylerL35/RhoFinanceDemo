# Pilot Demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A static, offline-capable, hosted MVP of Pilot that carries a 75-second pitch demo and lets a judge model their own company on their own phone.

**Architecture:** One `Scenario` object is the only mutable state. Four pure engine functions (`raiseBy`, `attribute`, `priceOption`, `alertsFor`) derive every displayed number as a `Derived` record with inputs and a formula. Nine hash-routed screens render those records and write only to `state`. No framework, no build step, no dependencies, no network at runtime.

**Tech Stack:** Plain ES modules in the browser (`<script type="module">`), `node --test` for tests, a service worker for offline, Vercel for static hosting.

**Spec:** `docs/superpowers/specs/2026-11-11-raise-by-demo-design.md` (rev 2). Read it alongside this plan — the plan implements it and every number below comes from it.

## Global Constraints

- **Zero runtime dependencies.** `package.json` exists only for `"type": "module"` and the test script. No `dependencies`, no `devDependencies`, no bundler.
- **No network at runtime.** No `fetch` outside `scripts/fetch-rho-sandbox.mjs`.
- **No `Date.now()` and no `new Date()` without arguments** anywhere in `public/js`. Time comes from `scenario.asOf`. Tests assert this by grep.
- **Money is integer cents** in the fixture and in `state`. Float cents inside the engine. Rounding happens only in `money.js` and `derived.js`.
- **Dates are UTC midnight** `Date` objects. Always `Date.UTC(...)`, never local constructors.
- **`MONTH_DAYS = 365.25 / 12`** (30.4375), exported from `calendar.js`, used only via `derived.js`.
- **`asOf` does not burn.** Accrual starts `asOf + 1 day`.
- **Copy is fixed.** Every user-facing string in tasks 7-14 is given verbatim. Do not paraphrase, do not add em dashes, do not add emoji.
- **Test mode throws:** `derived()` throws when `process.env.NODE_TEST_CONTEXT` is set (which `node --test` sets) and `inputs` is empty.
- **Palette:** abyss `#151716`, mint `#6AF3D9`, mint-deep `#08A68A`, moon `#FFFFFF`/`#F7F9F9`, card fill `#EFF3F2`, border `#D7E0DE`, muted text `#747D79`. 6px radius. Mint is the only accent.

```js
export function parseDate(iso) { return new Date(`${iso}T00:00:00Z`); }

export function addDays(d, n) { return new Date(d.getTime() + n * 86400000); }

export function diffDays(a, b) { return Math.round((b.getTime() - a.getTime()) / 86400000); }

export function daysInMonth(y, m) { return new Date(Date.UTC(y, m, 0)).getUTCDate(); }

export function monthStart(d) { return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), 1)); }

export function addMonths(d, n) {
  const total = d.getUTCFullYear() * 12 + d.getUTCMonth() + n;
  const y = Math.floor(total / 12);
  const m = total % 12;
  return new Date(Date.UTC(y, m, Math.min(d.getUTCDate(), daysInMonth(y, m + 1))));
}

}

export function fmtMonthDay(d) {
  return d.toLocaleString('en-US', { month: 'short', day: 'numeric', timeZone: 'UTC' });
```

```js
export function usd(cents) {
  return `$${Math.round(cents / 100).toLocaleString('en-US')}`;
}

export function shortUsd(cents) {
  const d = cents / 100;
  if (Math.abs(d) >= 1e6) return `$${(d / 1e6).toFixed(1)}M`;
  if (Math.abs(d) >= 1e3) return `$${Math.round(d / 1e3)}K`;
  return `$${Math.round(d)}`;
}

export function shortUsdSigned(cents) {
  const s = shortUsd(Math.abs(cents));
  return cents < 0 ? `-${s}` : `+${s}`;
}
```

```js
// public/js/derived.js
import { shortUsd } from './money.js';

const STRICT = typeof process !== 'undefined' && !!process.env.NODE_TEST_CONTEXT;

export function derived(value, unit, inputs, formula) {
  if (STRICT && (!inputs || inputs.length === 0)) {
    throw new Error(`no inputs for derived number: ${formula}`);
  }
  return { value, unit, inputs, formula };
}

export function fmt(d) {
  if (!d) return '—';
  if (!d.inputs || d.inputs.length === 0) {
    console.warn('[provenance] number without inputs:', d.formula);
    return '—';
  }
  switch (d.unit) {
    case 'cents': return shortUsd(d.value);
    case 'days': return `${Math.round(d.value)} days`;
    case 'months': return `${d.value.toFixed(1)} mo`;
    case 'count': return String(Math.round(d.value));
    default: return String(d.value);
  }
}

export function displayTenths(parts, totalMonths) {
  const target = Math.round(totalMonths * 10);
  const raw = parts.map((p) => p * 10);
  const out = raw.map((v) => Math.floor(v));
  let remaining = target - out.reduce((a, b) => a + b, 0);
  const order = raw
    .map((v, i) => ({ i, frac: v - Math.floor(v), v }))
    .sort((a, b) => b.frac - a.frac || b.v - a.v);
  let k = 0;
  while (remaining > 0) {
    out[order[k % order.length].i] += 1;
    remaining -= 1;
    k += 1;
  }
  return out;
}
```

```js
// public/js/scenario.js
import { parseDate, addDays, diffDays } from './calendar.js';

const KEY = 'raise-by.scenario.v1';

export async function loadFixture() {
  const res = await fetch('./data/fixture-seed.json');
  return res.json();
}

export function clone(sc) { return structuredClone(sc); }

export function validate(sc) {
  const errs = [];
  if (!(sc.cash > 0)) errs.push('NO_CASH');
  if (!(sc.plan.plannedGrossSpend >= 0)) errs.push('NO_SPEND');
  if (!(sc.revenue.current > 0)) errs.push('NO_REVENUE');
  if (sc.plan.plannedGrossSpend - sc.revenue.current <= 0) errs.push('NO_BURN');
  if (!(sc.cushionMonths >= 1)) errs.push('NO_CUSHION');
  return errs;
}

export function terms(sc, contractId, asOf = parseDate(sc.asOf)) {
  const c = sc.contracts.find((x) => x.id === contractId);
  const deadline = addDays(parseDate(c.renewsOn), -c.noticeDays);
  const daysLeft = diffDays(asOf, deadline);
  return { deadline, daysLeft, canCancel: daysLeft >= 0 };
}

export function saveLocal(sc) {
  if (globalThis.__RAISE_BY_PRESENT__) return;
  localStorage.setItem(KEY, JSON.stringify(sc));
}

export function loadLocal() {
  if (globalThis.__RAISE_BY_PRESENT__) return null;
  const raw = localStorage.getItem(KEY);
  return raw ? JSON.parse(raw) : null;
}

export function clearLocal() { localStorage.removeItem(KEY); }
```

```js
// public/js/engine.js
import { parseDate, addDays, diffDays, daysInMonth, monthStart, MONTH_DAYS } from './calendar.js';
import { derived } from './derived.js';

const HORIZON_MONTHS = 36;

export function payrollFor(sc, y, m) {
  let total = 0;
  for (const p of sc.payroll) {
    const s = parseDate(p.startsOn);
    const monthIndex = y * 12 + (m - 1);
    const startIndex = s.getUTCFullYear() * 12 + (s.getUTCMonth());
    if (monthIndex > startIndex) total += p.monthlyCost;
    else if (monthIndex === startIndex) {
      const dim = daysInMonth(y, m);
      total += (p.monthlyCost * (dim - s.getUTCDate() + 1)) / dim;
    }
  }
  return total;
}

function contractMonthly(sc, y, m) {
  const c = sc.contracts[0];
  if (c.choice !== 'monthly') return 0;
  const start = parseDate(c.renewsOn);
  const monthIndex = y * 12 + (m - 1);
  const startIndex = start.getUTCFullYear() * 12 + start.getUTCMonth();
  return monthIndex >= startIndex ? c.monthlyEquivalent : 0;
}

export function grossSpend(sc, y, m) {
  return sc.plan.plannedGrossSpend + payrollFor(sc, y, m) + contractMonthly(sc, y, m);
}

export function netBurn(sc, y, m) {
  const revenue = sc.revenue.current * (sc.revenue.priceMultiplier ?? 1);
  return grossSpend(sc, y, m) - revenue;
}

function oneOffsBetween(sc, from, to) {
  let delta = 0;
  const c = sc.contracts[0];
  const renews = parseDate(c.renewsOn);
  if (c.choice !== 'cancel' && c.choice !== 'monthly' && renews > from && renews <= to) {
    delta -= c.amount;
  }
  for (const inv of sc.invoices) {
    if (inv.collected) {
      const when = addDays(from, 1);
      if (when > from && when <= to) delta += inv.amount;
    }
  }
  return delta;
}

export function cashAt(sc, d) {
  const asOf = parseDate(sc.asOf);
  if (diffDays(asOf, d) <= 0) return sc.cash;
  let cash = sc.cash;
  for (let cursor = addDays(asOf, 1); cursor <= d; cursor = addDays(cursor, 1)) {
    const y = cursor.getUTCFullYear();
    const m = cursor.getUTCMonth() + 1;
    cash -= netBurn(sc, y, m) / daysInMonth(y, m);
  }
  return cash + oneOffsBetween(sc, asOf, d);
}

export function bufferAt(sc, d) {
  let total = 0;
  let cursor = monthStart(d);
  for (let i = 0; i < sc.cushionMonths; i++) {
    total += netBurn(sc, cursor.getUTCFullYear(), cursor.getUTCMonth() + 1);
    cursor = new Date(Date.UTC(cursor.getUTCFullYear(), cursor.getUTCMonth() + 1, 1));
  }
  return total;
}

export function raiseBy(sc) {
  const asOf = parseDate(sc.asOf);
  for (let i = 0; i <= HORIZON_MONTHS * 31; i++) {
    const d = addDays(asOf, i);
    if (cashAt(sc, d) <= bufferAt(sc, d)) return d;
  }
  return addDays(asOf, HORIZON_MONTHS * 31);
}

export function projectDate(sc) {
  const d = raiseBy(sc);
  return derived(d.getTime(), 'date', [
    { label: 'cash', value: sc.cash, source: 'scenario' },
    { label: 'net burn (Nov)', value: netBurn(sc, 2026, 11), source: 'engine' },
    { label: 'buffer', value: bufferAt(sc, parseDate(sc.asOf)), source: 'engine' },
    { label: 'cushion', value: sc.cushionMonths, source: 'scenario' },
  ], 'first day cash <= buffer; buffer = sum of next cushionMonths of netBurn');
}

export function revenueBand(sc) {
  const ratios = sc.revenue.trailing.map((t) => t.actual / t.expected);
  const mean = ratios.reduce((a, b) => a + b, 0) / ratios.length;
  const variance = ratios.reduce((a, r) => a + (r - mean) ** 2, 0) / (ratios.length - 1);
  const s = Math.sqrt(variance);
  const planned = sc.plan.plannedRevenue;
  return {
    mean, s,
    low: planned * Math.max(0.85, mean - s),
    high: planned * Math.min(1.05, mean + s),
  };
}

export function range(sc) {
  const band = revenueBand(sc);
  const at = (revenue) => {
    const c = structuredClone(sc);
    c.revenue.current = revenue;
    c.revenue.priceMultiplier = 1;
    return raiseBy(c);
  };
  return { low: at(band.low), base: raiseBy(sc), high: at(band.high), band };
}

export { MONTH_DAYS, diffDays, parseDate, addDays };
```

```js
function variantRemoving(sc, { signed, notCertain }) {
  const v = structuredClone(sc);
  const hiresInSigned = v.payroll.every((p) => p.confirmed);
  if (signed) {
    v.contracts[0].choice = 'cancel';
    if (hiresInSigned) v.payroll = [];

    v.revenue.current = v.plan.plannedRevenue;
    if (!hiresInSigned) v.payroll = [];

export function attribute(sc) {
  const hiresConfirmed = sc.payroll.length > 0 && sc.payroll.every((p) => p.confirmed);
  const current = raiseBy(sc);
  const signedOnly = raiseBy(variantRemoving(sc, { signed: true, notCertain: false }));
  const uncertainOnly = raiseBy(variantRemoving(sc, { signed: false, notCertain: true }));
  const anchor = raiseBy(variantRemoving(sc, { signed: true, notCertain: true }));

  const signedDays = ((signedOnly - current) / 86400000 + (anchor - uncertainOnly) / 86400000) / 2;
  const uncertainDays = ((uncertainOnly - current) / 86400000 + (anchor - signedOnly) / 86400000) / 2;
  const totalDays = (anchor - current) / 86400000;

  return {

    segments: [
      { id: 'signed', days: signedDays, months: signedDays / MONTH_DAYS },
      { id: 'notCertain', days: uncertainDays, months: uncertainDays / MONTH_DAYS },
```

```js
// public/js/intents.js
import { terms } from './scenario.js';
import { shortUsdSigned } from './money.js';

const MONTHS = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];
const FULL = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];

const HIRE = /^hire\s+(\d+)\s+(?:more\s+)?(?:engineers?|people|employees?)?(?:\s+in\s+(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)\w*)?\s*(2027)?$/i;
const LONGER = /(?:the\s+)?raise\s+takes\s+(\d+)\s+months?\s+longer/i;
const CUSHION = /^cushion\s+(\d+)\s+months?$/i;
const PRICES = /^(?:raise\s+prices|price\s+increase)\s+(\d+)\s*%$/i;
const CANCEL = /^cancel\s+(\w+)$/i;
const MONTHLY = /^switch\s+(\w+)\s+to\s+monthly$/i;

const refuse = (reason) => ({ ok: false, reason });

export function parseIntent(text, sc) {
  const t = text.trim().toLowerCase();

  let m = t.match(HIRE);
  if (m) {
    const n = Number(m[1]);
    const month = m[2] ? MONTHS.indexOf(m[2].slice(0, 3)) + 1 : null;
    const startMonth = month ?? 1;
    const year = m[3] ? 2027 : 2027;
    const cost = sc.payroll.at(-1)?.monthlyCost ?? 900000;
    const add = n * cost;
    return {
      ok: true,
      echo: `I read that as: ${n} more hires starting ${FULL[startMonth - 1]} ${year}, ${shortUsdSigned(add)}/mo.`,
      mutation: (v) => {
        for (let i = 0; i < n; i += 1) {
          v.payroll.push({ id: `whatif-${v.payroll.length + 1}`, label: `Extra hire ${i + 1}`,
            monthlyCost: cost, startsOn: `${year}-${String(startMonth).padStart(2, '0')}-01`, confirmed: true });
        }
      },
    };
  }

  m = t.match(LONGER);
  if (m) {
    const n = Number(m[1]);
    return { ok: true, echo: `I read that as: your raise takes ${n} months longer, so you need ${sc.cushionMonths + n} months of cushion. That moves your date earlier.`,
      mutation: (v) => { v.cushionMonths += n; } };
  }

  if (/^delay\s+the\s+raise/i.test(t)) {
    return refuse('That reads as "start later", which moves the date later. Try: "the raise takes 3 months longer".');
  }

  m = t.match(CUSHION);
  if (m) return { ok: true, echo: `I read that as: ${m[1]} months of cushion.`,
    mutation: (v) => { v.cushionMonths = Number(m[1]); } };

  m = t.match(PRICES);
  if (m) return { ok: true, echo: `I read that as: prices ${m[1]}% higher from next month, and on both ends of the range.`,
    mutation: (v) => { v.revenue.priceMultiplier = (v.revenue.priceMultiplier ?? 1) * (1 + Number(m[1]) / 100); } };

  m = t.match(CANCEL);
  if (m) {
    if (m[1] !== 'hubspot') return refuse(`I can't find a contract for "${m[1]}".`);
    const n = terms(sc, 'hubspot');
    if (!n.canCancel) return refuse('The notice window for HubSpot is closed, so it renews automatically.');
    return { ok: true, echo: 'I read that as: cancel the HubSpot renewal before the notice window closes.',
      mutation: (v) => { v.contracts[0].choice = 'cancel'; } };
  }

  m = t.match(MONTHLY);
  if (m) return { ok: true, echo: 'I read that as: switch HubSpot to monthly billing.',
    mutation: (v) => { v.contracts[0].choice = 'monthly'; } };

  return refuse("I can model hires, contract changes, price changes, and the cushion. I can't model that yet.");
}
```
